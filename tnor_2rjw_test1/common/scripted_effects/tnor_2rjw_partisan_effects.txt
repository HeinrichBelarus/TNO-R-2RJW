tnor_2rjw_open_partisan_supply = {
    set_country_flag = tnor_2rjw_partisan_supply_open
    set_variable = { tnor_2rjw_mongolia_divisions = 0 }
    set_variable = { tnor_2rjw_primorye_divisions = 0 }
}

# One brigade: five infantry battalions, anti-tank and engineer support.
tnor_2rjw_pay_partisan_supply_kit = {
    add_equipment_to_stockpile = { type = infantry_equipment amount = -615 }
    add_equipment_to_stockpile = { type = anti_tank_equipment amount = -72 }
    add_equipment_to_stockpile = { type = support_equipment amount = -30 }
}

tnor_2rjw_raise_irkutsk_infantry = {
    if = {
        limit = { controls_state = 566 }
        tnor_2rjw_ensure_rifle_template = yes
        566 = {
            create_unit = {
                division = "division_template = \"Eastern Rifle Division\" start_experience_factor = 1 start_equipment_factor = 1"
                owner = ROOT
                prioritize_location = 1727
                count = 2
            }
        }
    }
}

tnor_2rjw_ensure_rifle_template = {
    if = {
        limit = { NOT = { has_template = "Eastern Rifle Division" } }
        division_template = {
            name = "Eastern Rifle Division"
            regiments = {
                infantry = { x = 0 y = 0 }
                infantry = { x = 0 y = 1 }
                infantry = { x = 0 y = 2 }
                infantry = { x = 1 y = 0 }
                infantry = { x = 1 y = 1 }
                infantry = { x = 1 y = 2 }
            }
        }
    }
}

tnor_2rjw_prepare_partisan_zones = {
    every_state = {
        limit = {
            NOT = { is_demilitarized_zone = yes }
            OR = {
                AND = {
                    is_owned_by = MEN
                    OR = { state = 814 state = 815 }
                }
                AND = {
                    tnor_2rjw_far_eastern_partisan_land = yes
                    NOT = { state = 1646 }
                }
            }
        }
        set_demilitarized_zone = yes
        set_state_flag = tnor_2rjw_partisan_dmz
    }
}

tnor_2rjw_establish_partisan_states = {
    every_state = {
        limit = { tnor_2rjw_mongolian_partisan_land = yes }
        add_core_of = MPS
        MPS = { transfer_state = PREV }
    }
    every_state = {
        limit = { tnor_2rjw_far_eastern_partisan_land = yes }
        add_core_of = FPS
        FPS = { transfer_state = PREV }
    }
    if = {
        limit = { country_exists = MPS }
        MPS = {
            if = {
                limit = { 330 = { is_owned_by = PREV } }
                set_capital = { state = 330 }
            }
            tnor_2rjw_prepare_partisan_country = yes
        }
    }
    if = {
        limit = { country_exists = FPS }
        FPS = {
            if = {
                limit = { 1646 = { is_owned_by = PREV } }
                set_capital = { state = 1646 }
            }
            tnor_2rjw_prepare_partisan_country = yes
        }
    }
}

# Called in the newly formed partisan country's scope.
tnor_2rjw_prepare_partisan_country = {
    inherit_technology = ROOT
    division_template = {
        name = "Partisan Brigade"
        regiments = {
            infantry = { x = 0 y = 0 }
            infantry = { x = 0 y = 1 }
            infantry = { x = 0 y = 2 }
            infantry = { x = 1 y = 0 }
            infantry = { x = 1 y = 1 }
        }
        support = {
            anti_tank = { x = 0 y = 0 }
            engineer = { x = 0 y = 1 }
        }
    }
}

tnor_2rjw_deploy_partisans = {
    set_country_flag = tnor_2rjw_partisan_supply_closed
    if = {
        limit = { country_exists = MPS check_variable = { tnor_2rjw_mongolia_divisions > 0 } }
        set_temp_variable = { tnor_2rjw_units_to_deploy = tnor_2rjw_mongolia_divisions }
        MPS = { tnor_2rjw_deploy_partisan_country = yes }
    }
    if = {
        limit = { country_exists = FPS check_variable = { tnor_2rjw_primorye_divisions > 0 } }
        set_temp_variable = { tnor_2rjw_units_to_deploy = tnor_2rjw_primorye_divisions }
        FPS = { tnor_2rjw_deploy_partisan_country = yes }
    }
}

tnor_2rjw_deploy_partisan_country = {
    capital_scope = {
        # If an assembly area is still demilitarised, open it for deployment.
        if = {
            limit = { has_state_flag = tnor_2rjw_partisan_dmz }
            set_demilitarized_zone = no
            clr_state_flag = tnor_2rjw_partisan_dmz
        }
        create_unit = {
            division = "division_template = \"Partisan Brigade\" start_equipment_factor = 1"
            owner = PREV
            count = tnor_2rjw_units_to_deploy
        }
    }
}

tnor_2rjw_declare_eastern_war = {
    set_country_flag = tnor_2rjw_eastern_war_started
    every_state = {
        limit = { has_state_flag = tnor_2rjw_partisan_dmz }
        set_demilitarized_zone = no
        clr_state_flag = tnor_2rjw_partisan_dmz
    }
    if = {
        limit = { is_in_faction = no }
        create_faction_from_template = {
            template = faction_template_generic
            name = collective_security_treaty_organization
            icon = GFX_collective_security_treaty_organization
        }
        add_ideas = CSTO_Guardian_of_Eurasia
    }
    every_country = {
        limit = {
            exists = yes
            OR = { tag = MPS tag = FPS }
            NOT = { is_in_faction_with = ROOT }
        }
        ROOT = { add_to_faction = PREV }
        add_ideas = CSTO_Member
    }
    if = {
        limit = {
            has_completed_focus = TNOR_2RJW_the_tiger_and_the_bear
            country_exists = IND
            country_exists = AZH
        }
        AZH = {
            tnor_2rjw_detach_dragon_target = yes
            econ_leave_sphere = yes
        }
        IND = {
            if = {
                limit = { NOT = { has_war_with = AZH } }
                declare_war_on = { target = AZH type = annex_everything }
            }
        }
    }
    every_country = {
        limit = {
            exists = yes
            OR = { tag = JAP tag = MEN tag = MAN }
            NOT = { is_in_faction_with = ROOT }
        }
        ROOT = {
            if = {
                limit = { NOT = { has_war_with = PREV } }
                declare_war_on = { target = PREV type = annex_everything }
            }
        }
        if = {
            limit = { has_war_with = ROOT }
            if = {
                limit = { country_exists = MPS }
                MPS = { tnor_2rjw_join_eastern_front = yes }
            }
            if = {
                limit = { country_exists = FPS }
                FPS = { tnor_2rjw_join_eastern_front = yes }
            }
        }
    }
}

# Called in a partisan country's scope, with the opposing country as PREV.
tnor_2rjw_join_eastern_front = {
    if = {
        limit = { NOT = { has_war_with = PREV } }
        add_to_war = { targeted_alliance = ROOT enemy = PREV }
    }
}
