# Called in the target country's scope. Leave any existing patron's bloc first.
tnor_2rjw_detach_dragon_target = {
    if = {
        limit = { is_in_faction = yes }
        if = {
            limit = { is_faction_leader = yes }
            dismantle_faction = yes
        }
        else = { leave_faction = yes }
    }
    if = {
        limit = { is_subject = yes }
        OVERLORD = { end_puppet = PREV }
    }
}

tnor_2rjw_make_dragon_subject = {
    tnor_2rjw_detach_dragon_target = yes
    event_target:tnor_2rjw_campaign_russia = { puppet = PREV }
}

tnor_2rjw_send_dragon_proposals = {
    every_country = {
        limit = {
            exists = yes
            OR = { tag = XIN tag = QIN }
            NOT = { is_subject_of = ROOT }
        }
        country_event = { id = tnor_2rjw_dragon.2 }
    }
}

tnor_2rjw_launch_broken_dragon = {
    every_country = {
        limit = {
            exists = yes
            OR = { tag = XIN tag = QIN }
            NOT = { is_subject_of = ROOT }
        }
        tnor_2rjw_detach_dragon_target = yes
        set_country_flag = tnor_2rjw_dragon_war_target
        ROOT = {
            if = {
                limit = { NOT = { has_war_with = PREV } }
                declare_war_on = { target = PREV type = puppet_wargoal_focus }
            }
        }
        country_event = { id = tnor_2rjw_dragon.6 }
    }
}

tnor_2rjw_ratify_dragon_agreements = {
    every_country = {
        limit = {
            exists = yes
            OR = { tag = XIN tag = QIN }
            has_country_flag = tnor_2rjw_dragon_accepted
            NOT = { is_subject_of = ROOT }
        }
        clr_country_flag = tnor_2rjw_dragon_accepted
        tnor_2rjw_make_dragon_subject = yes
    }
}

tnor_2rjw_handle_dragon_capitulation = {
    ROOT = {
        clr_country_flag = tnor_2rjw_dragon_war_target
        tnor_2rjw_make_dragon_subject = yes
    }
    event_target:tnor_2rjw_campaign_russia = { country_event = { id = tnor_2rjw_dragon.4 } }
}
