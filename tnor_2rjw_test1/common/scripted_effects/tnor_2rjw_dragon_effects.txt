# Called in the target country's scope. Leave any existing patron's bloc first.
tnor_2rjw_detach_dragon_target = {
    if = {
        limit = { is_in_faction = yes }
        if = {
            limit = { is_faction_leader = yes }
            dismantle_faction = yes
        }
        else = { leave_faction = yes }
    }
    if = {
        limit = { is_subject = yes }
        OVERLORD = { end_puppet = PREV }
    }
}

tnor_2rjw_make_dragon_subject = {
    tnor_2rjw_detach_dragon_target = yes
    event_target:tnor_2rjw_campaign_russia = { puppet = PREV }
}

tnor_2rjw_send_dragon_proposals = {
    every_country = {
        limit = {
            exists = yes
            OR = { tag = XIN tag = QIN }
        }
        country_event = { id = tnor_2rjw_dragon.2 }
    }
}

tnor_2rjw_launch_broken_dragon = {
    every_country = {
        limit = {
            exists = yes
            OR = { tag = XIN tag = QIN }
            NOT = { is_subject_of = ROOT }
        }
        tnor_2rjw_detach_dragon_target = yes
        ROOT = {
            if = {
                limit = { NOT = { has_war_with = PREV } }
                declare_war_on = { target = PREV type = annex_everything }
            }
        }
        country_event = { id = tnor_2rjw_dragon.6 }
    }
}

tnor_2rjw_settle_broken_dragon = {
    if = {
        limit = { has_completed_focus = TNOR_2RJW_operation_broken_dragon }
        XIN = {
            transfer_state = 618
            transfer_state = 617
            transfer_state = 957
            transfer_state = 958
            transfer_state = 619
            transfer_state = 1964
        }
        QIN = {
            transfer_state = 764
            transfer_state = 604
            transfer_state = 906
        }
    }
    save_global_event_target_as = tnor_2rjw_campaign_russia
    XIN = {
        if = {
            limit = { NOT = { is_subject_of = ROOT } }
            tnor_2rjw_make_dragon_subject = yes
        }
    }
    QIN = {
        if = {
            limit = { NOT = { is_subject_of = ROOT } }
            tnor_2rjw_make_dragon_subject = yes
        }
    }
    set_country_flag = tnor_2rjw_dragon_settled
    country_event = { id = tnor_2rjw_dragon.4 }
}
