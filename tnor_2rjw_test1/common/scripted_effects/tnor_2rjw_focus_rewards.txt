# Build a hub at a town where possible, then connect it to the closest existing
# railhead or port. Do not duplicate a hub that the player has already built.
tnor_2rjw_build_border_supply_hubs = {
    custom_effect_tooltip = tnor_2rjw_border_supply_hubs_tt
    hidden_effect = {
        if = {
            limit = { OR = { country_exists = MAN country_exists = MEN } }
            every_owned_state = {
                limit = {
                    is_controlled_by = ROOT
                    any_neighbor_state = {
                        owner = {
                            OR = { tag = MAN tag = MEN }
                            exists = yes
                        }
                    }
                    NOT = { tnor_2rjw_has_supply_hub = yes }
                }
                add_building_construction = {
                    type = supply_node
                    level = 1
                    instant_build = yes
                    province = { all_provinces = yes limit_to_victory_point = yes first = yes }
                }
                if = {
                    limit = { NOT = { tnor_2rjw_has_supply_hub = yes } }
                    add_building_construction = {
                        type = supply_node
                        level = 1
                        instant_build = yes
                        province = { all_provinces = yes first = yes }
                    }
                }
                if = {
                    limit = { tnor_2rjw_has_supply_hub = yes }
                    tnor_2rjw_link_new_border_hub = yes
                }
            }
        }
    }
}

tnor_2rjw_link_new_border_hub = {
    save_event_target_as = tnor_2rjw_border_hub
    set_temp_variable = { tnor_2rjw_nearest_railhead_distance = 100000 }
    ROOT = {
        every_owned_state = {
            limit = {
                NOT = { state = event_target:tnor_2rjw_border_hub }
                NOT = { has_state_flag = tnor_2rjw_recent_border_hub }
                is_controlled_by = ROOT
                OR = {
                    tnor_2rjw_has_supply_hub = yes
                    any_province_building_level = {
                        building = naval_base
                        level > 0
                        province = { all_provinces = yes }
                    }
                }
            }
            event_target:tnor_2rjw_border_hub = {
                set_temp_variable = { tnor_2rjw_candidate_distance = distance_to@PREV }
            }
            if = {
                limit = {
                    check_variable = { tnor_2rjw_candidate_distance < tnor_2rjw_nearest_railhead_distance }
                    ROOT = {
                        can_build_railway = {
                            start_state = PREV
                            target_state = event_target:tnor_2rjw_border_hub
                            build_only_on_allied = yes
                        }
                    }
                }
                set_temp_variable = { tnor_2rjw_nearest_railhead_distance = tnor_2rjw_candidate_distance }
                save_event_target_as = tnor_2rjw_nearest_railhead
            }
        }
        if = {
            limit = { check_variable = { tnor_2rjw_nearest_railhead_distance < 100000 } }
            build_railway = {
                start_state = event_target:tnor_2rjw_nearest_railhead
                target_state = event_target:tnor_2rjw_border_hub
                level = 1
                build_only_on_allied = yes
            }
        }
    }
    set_state_flag = { flag = tnor_2rjw_recent_border_hub days = 1 }
}

# The Omsk tree uses the same border selector for the Moscow frontier.
tnor_2rjw_fortify_german_border = {
    every_owned_state = {

            level = 2
            instant_build = yes

                all_provinces = yes

        limit = { any_neighbor_state = { is_owned_by = MCW } }

                limit_to_border_country = MCW

            }

}
